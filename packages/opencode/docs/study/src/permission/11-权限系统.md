# 权限系统

权限系统是 OpenCode 实现细粒度访问控制的核心机制，控制 AI 可以执行的操作。

## 核心文件

- `src/permission/next.ts` - 权限系统实现

## 权限类型

OpenCode 支持多种权限类型，对应不同的操作：

```typescript
type Permission =
  | "bash"           // 执行 shell 命令
  | "edit"           // 编辑文件
  | "external_directory"  // 访问外部目录
  | "glob"           // 文件匹配
  | "grep"           // 内容搜索
  | "list"           // 列出目录
  | "question"       // 向用户提问
  | "read"           // 读取文件
  | "task"           // 调用子代理
  | "todoread"       // 读取 TODO
  | "todowrite"      // 写入 TODO
  | "webfetch"       // 获取网页
  | "websearch"      // 网页搜索
  | "codesearch"     // 代码搜索
  | "plan_enter"     // 进入计划模式
  | "plan_exit"      // 退出计划模式
  | "doom_loop"      // 死循环检测
```

## 权限规则结构

```typescript
export type Action = "allow" | "deny" | "ask"
export type Pattern = string | Record<string, Action>

export const Rule = z.object({
  permission: z.string(),                  // 权限类型
  action: Action,                          // 操作
  pattern: Pattern,                        // 模式
})

export type Rule = z.infer<typeof Rule>
export type Ruleset = Rule[]
```

## 权限配置格式

权限可以从配置对象创建：

```typescript
// 简单格式
PermissionNext.fromConfig({
  "bash": "allow",     // 允许所有 bash 操作
  "edit": "deny",      // 禁止所有编辑操作
  "read": "ask",       // 询问用户
})

// 嵌套格式（针对文件路径）
PermissionNext.fromConfig({
  "read": {
    "*": "allow",              // 允许读取所有文件
    "*.env": "ask",            // 询问 .env 文件
    "*.env.*": "ask",          // 询问 .env.* 文件
    "*.env.example": "allow",  // 允许 .env.example
  },
  "edit": {
    "*": "deny",               // 禁止编辑所有文件
    ".opencode/**": "allow",   // 允许编辑 .opencode 目录
  },
})
```

## 权限评估

```typescript
export function evaluate(
  permission: string,
  pattern: string,
  ruleset: Ruleset
): { action: Action; rule?: Rule } {
  // 1. 查找匹配的规则
  const rules = ruleset.filter((r) => r.permission === permission)

  // 2. 精确匹配优先
  const exact = rules.find((r) => r.pattern === pattern)
  if (exact) return { action: exact.action, rule: exact }

  // 3. 通配符匹配
  for (const rule of rules) {
    if (typeof rule.pattern === "string" && wildcard(rule.pattern, pattern)) {
      return { action: rule.action, rule }
    }
  }

  // 4. 嵌套模式匹配
  for (const rule of rules) {
    if (typeof rule.pattern === "object") {
      for (const [key, action] of Object.entries(rule.pattern)) {
        if (wildcard(key, pattern)) {
          return { action, rule }
        }
      }
    }
  }

  // 5. 默认拒绝
  return { action: "deny" }
}
```

## 权限请求

```typescript
export async function ask(input: {
  permission: string
  patterns: string[]
  sessionID: string
  always?: string[]
  metadata?: Record<string, any>
  ruleset: Ruleset
}): Promise<void> {
  const { permission, patterns, sessionID, always = [], metadata, ruleset } = input

  // 1. 检查是否在 always 列表中（跳过权限检查）
  for (const pattern of patterns) {
    for (const alwaysPattern of always) {
      if (wildcard(alwaysPattern, pattern)) {
        return // 允许，不询问
      }
    }
  }

  // 2. 评估所有模式
  const results = patterns.map((pattern) =>
    evaluate(permission, pattern, ruleset)
  )

  // 3. 如果所有都是 allow，直接返回
  if (results.every((r) => r.action === "allow")) {
    return
  }

  // 4. 如果有 deny，抛出错误
  if (results.some((r) => r.action === "deny")) {
    throw new RejectedError(
      {
        permission,
        patterns,
        reason: "Permission denied by ruleset",
      },
      { cause: null }
    )
  }

  // 5. 如果有 ask，发送权限请求事件
  if (results.some((r) => r.action === "ask")) {
    const id = Identifier.ascending("permission")

    const request: Omit<Request, "id"> = {
      permission,
      patterns,
      sessionID,
      metadata,
    }

    // 发布事件，等待用户响应
    Bus.publish(Event.Asked, { id, ...request })

    // 等待响应
    const response = await new Promise<Approval | Rejection>((resolve) => {
      const unsub = Bus.subscribe(Event.Responded, (evt) => {
        if (evt.properties.requestID !== id) return
        unsub()
        resolve(evt.properties)
      })

      // 设置超时
      setTimeout(() => {
        unsub()
        resolve({ type: "reject", requestID: id })
      }, PERMISSION_TIMEOUT)
    })

    if (response.type === "reject") {
      throw new RejectedError(
        { permission, patterns, reason: "Permission rejected by user" },
        { cause: null }
      )
    }
  }
}
```

## 权限合并

```typescript
export function merge(...rulesets: Ruleset[]): Ruleset {
  const result: Ruleset = []

  for (const ruleset of rulesets) {
    for (const rule of ruleset) {
      // 查找是否已存在相同权限和模式的规则
      const existing = result.find(
        (r) => r.permission === rule.permission && r.pattern === rule.pattern
      )

      if (existing) {
        // 后面的规则覆盖前面的
        existing.action = rule.action
      } else {
        result.push({ ...rule })
      }
    }
  }

  return result
}
```

## 获取禁用的工具

```typescript
export function disabled(
  tools: string[],
  ruleset: Ruleset
): Set<string> {
  const disabled = new Set<string>()

  for (const tool of tools) {
    const result = evaluate(tool, "*", ruleset)
    if (result.action === "deny") {
      disabled.add(tool)
    }
  }

  return disabled
}
```

## 权限错误

```typescript
export class RejectedError extends NamedError.createClass(
  "PermissionRejectedError",
  z.object({
    permission: z.string(),
    patterns: z.string().array(),
    reason: z.string().optional(),
  })
) {
  constructor(
    input: { permission: string; patterns: string[]; reason?: string },
    cause?: Error | null
  ) {
    super(input, { cause })
  }
}
```

## 使用示例

### 在工具中请求权限

```typescript
execute: async (args, ctx) => {
  // 请求权限
  await ctx.ask({
    permission: "bash",
    patterns: [args.command],
    always: ["echo", "pwd", "ls"],  // 这些命令不需要权限检查
    metadata: { command: args.command },
  })

  // 执行操作
  const result = await executeCommand(args.command)
  return { title: "Done", output: result, metadata: {} }
}
```
