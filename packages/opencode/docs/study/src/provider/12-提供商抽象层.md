# 提供商抽象层

提供商抽象层统一支持 20+ AI 提供商。

## 核心接口

```typescript
export interface Provider {
  id: string              // 提供商 ID
  name: string            // 提供商名称
  options?: Record<string, any>  // 提供商选项
  models: Model[]         // 支持的模型
}

export interface Model {
  id: string              // 模型 ID
  providerID: string      // 提供商 ID
  api: {
    id: string            // API ID（用于 ai-sdk）
    npm: string           // npm 包名
  }
  capabilities: {
    vision?: boolean      // 支持视觉
    temperature?: boolean // 支持温度
    streaming?: boolean   // 支持流式
  }
  cost?: {
    input: number         // 输入价格（每百万 token）
    output: number        // 输出价格
    cache?: {
      read: number        // 缓存读取价格
      write: number       // 缓存写入价格
    }
  }
  limit: {
    input: number         // 输入 token 限制
    output: number        // 输出 token 限制
  }
  options?: Record<string, any>
}
```

## 获取提供商

```typescript
const provider = await Provider.getProvider("anthropic")
// {
//   id: "anthropic",
//   name: "Anthropic",
//   models: [...]
// }
```

## 获取模型

```typescript
const model = await Provider.getModel("anthropic", "claude-sonnet-4-20250514")
// {
//   id: "claude-sonnet-4-20250514",
//   providerID: "anthropic",
//   capabilities: { streaming: true, ... },
//   cost: { input: 3.0, output: 15.0 },
//   ...
// }
```

## 获取语言模型

```typescript
const language = await Provider.getLanguage(model)
// 返回 ai-sdk 可用的语言模型
```

## 参数转换

```typescript
export namespace ProviderTransform {
  // 转换选项
  function options(input: {
    model: Model
    sessionID: string
    providerOptions?: Record<string, any>
  }): Record<string, any>

  // 转换消息
  function message(
    messages: ModelMessage[],
    model: Model,
    options: Record<string, any>
  ): ModelMessage[]

  // 转换错误
  function error(providerID: string, error: APICallError): string

  // 转换提供商选项
  function providerOptions(
    model: Model,
    options: Record<string, any>
  ): Record<string, any>
}
```
