# 会话存储机制

## 概述

每次与 LLM 的交互（用户输入和 AI 输出）都会持久化存储在本地文件系统中。这使得会话可以被恢复、回顾和分析。

---

## 是否本地存储？

**是的**，所有交互数据都存储在本地，包括：

1. **会话元数据** - 会话ID、标题、时间戳、父子关系等
2. **用户消息** - 用户的输入内容
3. **助手消息** - AI 的响应内容
4. **消息部分** - 文本、工具调用、文件附件等
5. **使用量统计** - Token 使用量、成本计算

### 存储层 API

```typescript
// src/storage/storage.ts - 存储层核心API

// 写入数据
export async function write<T>(key: string[], content: T) {
  const dir = await state().then((x) => x.dir)
  const target = path.join(dir, ...key) + ".json"
  await Bun.write(target, JSON.stringify(content, null, 2))
}

// 读取数据
export async function read<T>(key: string[]) {
  const dir = await state().then((x) => x.dir)
  const target = path.join(dir, ...key) + ".json"
  return await Bun.file(target).json() as T
}

// 更新数据
export async function update<T>(key: string[], fn: (draft: T) => void) {
  const target = path.join(dir, ...key) + ".json"
  const content = await Bun.file(target).json()
  fn(content)
  await Bun.write(target, JSON.stringify(content, null, 2))
}

// 删除数据
export async function remove(key: string[]) {
  const target = path.join(dir, ...key) + ".json"
  await fs.unlink(target).catch(() => {})
}

// 列出目录
export async function list(prefix: string[]) {
  const dir = path.join(dir, ...prefix)
  return await Array.fromAsync(
    glob.scan({ cwd: dir, onlyFiles: true })
  )
}
```

## 会话存储操作

### 创建会话

```typescript
// src/session/index.ts:206-247
export async function createNext(input: {
  id?: string
  title?: string
  parentID?: string
  directory: string
  permission?: PermissionNext.Ruleset
}) {
  const result: Info = {
    id: Identifier.descending("session", input.id),
    slug: Slug.create(),
    version: Installation.VERSION,
    projectID: Instance.project.id,
    directory: input.directory,
    parentID: input.parentID,
    title: input.title ?? createDefaultTitle(!!input.parentID),
    permission: input.permission,
    time: {
      created: Date.now(),
      updated: Date.now(),
    },
  }
  // 存储会话元数据
  await Storage.write(["session", Instance.project.id, result.id], result)
  Bus.publish(Event.Created, { info: result })
  return result
}
```

### 获取会话的所有消息

```typescript
// src/session/index.ts:316-330
export const messages = fn(
  z.object({
    sessionID: Identifier.schema("session"),
    limit: z.number().optional(),
  }),
  async (input) => {
    const result = [] as MessageV2.WithParts[]
    for await (const msg of MessageV2.stream(input.sessionID)) {
      if (input.limit && result.length >= input.limit) break
      result.push(msg)
    }
    result.reverse()  // 最新的消息在前
    return result
  },
)

// 使用示例
const msgs = await Session.messages({ sessionID: "session_abc123" })
```

### MessageV2.stream 实现

```typescript
// src/session/message-v2.ts - 消息流式读取
export async function* stream(sessionID: string): AsyncGenerator<WithParts> {
  for await (const item of Storage.list(["message", sessionID])) {
    const msgID = item.at(-1)!
    const info = await Storage.read<Info>(["message", sessionID, msgID])
    const parts: Part[] = []
    for await (const partItem of Storage.list(["part", msgID])) {
      const partID = partItem.at(-1)!
      const part = await Storage.read<Part>(["part", msgID, partID])
      parts.push(part)
    }
    yield { info, parts }
  }
}
```

## 完整示例：读取会话数据

```typescript
// 1. 获取会话信息
const session = await Session.get("session_abc123")
console.log("会话标题:", session.title)
console.log("会话时间:", new Date(session.time.created))

// 2. 获取会话的所有消息
const msgs = await Session.messages({ sessionID: "session_abc123" })

// 3. 遍历消息查看输入输出
for (const msg of msgs) {
  if (msg.info.role === "user") {
    // === 用户输入 ===
    console.log("\n=== 用户消息 ===")
    console.log("时间:", new Date(msg.info.time.created))
    console.log("代理:", msg.info.agent)
    console.log("模型:", msg.info.model.providerID + "/" + msg.info.model.modelID)

  } else if (msg.info.role === "assistant") {
    // === AI 输出 ===
    console.log("\n=== AI 消息 ===")
    console.log("时间:", new Date(msg.info.time.created))
    console.log("成本:", msg.info.cost, "USD")
    console.log("Token:", msg.info.tokens)

    // 查看 AI 的各个部分
    for (const part of msg.parts) {
      switch (part.type) {
        case "text":
          console.log("  [文本]", part.text)
          break

        case "tool":
          if (part.state.status === "completed") {
            console.log("  [工具]", part.tool)
            console.log("    标题:", part.state.title)
            console.log("    输出:", part.state.output.substring(0, 100) + "...")
          }
          break

        case "step-finish":
          console.log("  [步骤完成]")
          console.log("    原因:", part.reason)
          console.log("    Token:", part.tokens)
          console.log("    成本:", part.cost)
          break

        case "reasoning":
          console.log("  [推理]", part.text.substring(0, 100) + "...")
          break
      }
    }
  }
}
```
