# 工具系统

工具系统是 OpenCode 实现 AI 代码助手功能的核心机制之一。工具（Tool）是 AI 可以调用的函数，用于执行文件操作、代码搜索、命令执行等任务。

## 工具定义模式

### Tool.Info 接口

```typescript
export interface Info<
  Parameters extends z.ZodType = z.ZodType,
  M extends Metadata = Metadata
> {
  id: string                                          // 工具 ID
  init: (ctx?: InitContext) => Promise<{
    description: string                               // 工具描述（给 AI 看）
    parameters: Parameters                            // Zod 参数 schema
    execute(
      args: z.infer<Parameters>,
      ctx: Context,
    ): Promise<{
      title: string                                   // 结果标题
      metadata: M                                     // 元数据
      output: string                                  // 输出内容
      attachments?: MessageV2.FilePart[]             // 附件
    }>
    formatValidationError?(error: z.ZodError): string
  }>
}
```

### 工具定义示例

```typescript
export const ReadTool = Tool.define("read", {
  description: "Read the contents of a file",
  parameters: z.object({
    filePath: z.string().describe("The path to the file to read"),
    offset: z.number().optional().describe("The line number to start reading from"),
    limit: z.number().optional().describe("The number of lines to read"),
  }),
  execute: async (args, ctx) => {
    if (ctx.abort.signal.aborted) {
      throw new Error("Operation aborted")
    }

    const content = await fs.readFile(args.filePath, "utf-8")
    const lines = content.split("\n")

    let output = lines
    if (args.offset !== undefined) output = output.slice(args.offset - 1)
    if (args.limit !== undefined) output = output.slice(0, args.limit)

    return {
      title: args.filePath,
      output: output.join("\n"),
      metadata: { count: output.length },
    }
  },
})
```

### Tool.define() 函数

```typescript
export function define<Parameters extends z.ZodType, Result extends Metadata>(
  id: string,
  init: Info<Parameters, Result>["init"] | Awaited<ReturnType<Info<Parameters, Result>["init"]>>,
): Info<Parameters, Result> {
  return {
    id,
    init: async (initCtx) => {
      const toolInfo = init instanceof Function ? await init(initCtx) : init
      const execute = toolInfo.execute

      // 包装 execute 函数以提供通用功能
      toolInfo.execute = async (args, ctx) => {
        // 1. 参数验证
        try {
          toolInfo.parameters.parse(args)
        } catch (error) {
          if (error instanceof z.ZodError && toolInfo.formatValidationError) {
            throw new Error(toolInfo.formatValidationError(error), { cause: error })
          }
          throw new Error(
            `The ${id} tool was called with invalid arguments: ${error}.`,
            { cause: error },
          )
        }

        // 2. 执行原始函数
        const result = await execute(args, ctx)

        // 3. 输出截断处理
        if (result.metadata.truncated !== undefined) {
          return result
        }
        const truncated = await Truncate.output(result.output, {}, initCtx?.agent)
        return {
          ...result,
          output: truncated.content,
          metadata: {
            ...result.metadata,
            truncated: truncated.truncated,
            ...(truncated.truncated && { outputPath: truncated.outputPath }),
          },
        }
      }
      return toolInfo
    },
  }
}
```

## 工具执行上下文

```typescript
export type Context<M extends Metadata = Metadata> = {
  sessionID: string                // 会话 ID
  messageID: string                // 消息 ID
  agent: string                    // 代理名称
  abort: AbortSignal               // 中止信号
  callID?: string                  // 调用 ID（LLM 生成）
  extra?: { [key: string]: any }   // 额外数据
  messages: MessageV2.WithParts[]  // 消息历史
  metadata(input: {                // 更新元数据
    title?: string
    metadata?: M
  }): void
  ask(input: Omit<PermissionNext.Request, "id" | "sessionID" | "tool">): Promise<void>
}
```

## 核心工具详解

### 1. Read - 读取文件

```typescript
export const ReadTool = Tool.define("read", {
  description: "Read the contents of a file",
  parameters: z.object({
    filePath: z.string().describe("The path to the file to read"),
    offset: z.number().optional().describe("The line number to start reading from"),
    limit: z.number().optional().describe("The number of lines to read"),
  }),
  execute: async (args, ctx) => {
    // 1. 路径解析
    const absolutePath = path.resolve(Instance.worktree, args.filePath)

    // 2. 权限检查
    await ctx.ask({
      permission: "read",
      patterns: [args.filePath],
    })

    // 3. 读取文件
    const content = await Bun.file(absolutePath).text()
    const lines = content.split("\n")

    // 4. 应用范围
    let output = lines
    if (args.offset !== undefined) output = output.slice(args.offset - 1)
    if (args.limit !== undefined) output = output.slice(0, args.limit)

    // 5. 返回结果
    return {
      title: args.filePath,
      output: output.join("\n"),
      metadata: { count: output.length },
    }
  },
})
```

### 3. Bash - 执行命令

```typescript
export const BashTool = Tool.define("bash", {
  description: "Execute a shell command",
  parameters: z.object({
    command: z.string().describe("The command to execute"),
  }),
  execute: async (args, ctx) => {
    await ctx.ask({
      permission: "bash",
      patterns: [args.command],
      always: ["echo", "cd", "pwd", "ls"],
    })

    const proc = Bun.spawn(["sh", "-c", args.command], {
      cwd: Instance.worktree,
      stdout: "pipe",
      stderr: "pipe",
    })

    const output = await new Response(proc.stdout).text()
    const error = await new Response(proc.stderr).text()
    const exitCode = await proc.exited

    if (exitCode !== 0) {
      throw new Error(`Command failed with exit code ${exitCode}: ${error}`)
    }

    return {
      title: args.command,
      output: output.trim(),
      metadata: { exitCode },
    }
  },
})
```

### 4. Task - 子代理任务

这是最复杂的工具之一，用于创建子会话并委托给子代理执行任务。

```typescript
export const TaskTool = Tool.define("task", {
  description: DESCRIPTION,  // 动态生成，包含可用代理列表
  parameters: z.object({
    description: z.string().describe("A short (3-5 words) description of the task"),
    prompt: z.string().describe("The task for the agent to perform"),
    subagent_type: z.string().describe("The type of specialized agent to use"),
    session_id: z.string().describe("Existing Task session to continue").optional(),
  }),
  execute: async (params, ctx) => {
    // 1. 权限检查
    await ctx.ask({
      permission: "task",
      patterns: [params.subagent_type],
      always: ["*"],
    })

    // 2. 获取代理
    const agent = await Agent.get(params.subagent_type)
    if (!agent) throw new Error(`Unknown agent: ${params.subagent_type}`)

    // 3. 创建子会话
    const session = await Session.create({
      parentID: ctx.sessionID,
      title: params.description + ` (@${agent.name} subagent)`,
      permission: [/* 限制子代理的权限 */],
    })

    // 4. 监听子会话事件
    const parts: Record<string, {...}> = {}
    const unsub = Bus.subscribe(MessageV2.Event.PartUpdated, async (evt) => {
      if (evt.properties.part.sessionID !== session.id) return
      if (evt.properties.part.messageID === messageID) return
      if (evt.properties.part.type !== "tool") return

      const part = evt.properties.part
      parts[part.id] = {
        id: part.id,
        tool: part.tool,
        state: {
          status: part.state.status,
          title: part.state.status === "completed" ? part.state.title : undefined,
        },
      }
      ctx.metadata({ title: params.description, metadata: { ... } })
    })

    // 5. 发送提示词
    const result = await SessionPrompt.prompt({
      messageID,
      sessionID: session.id,
      model: agent.model ?? { /* 使用父会话模型 */ },
      agent: agent.name,
      tools: { /* 禁用某些工具 */ },
      parts: [{ type: "text", text: params.prompt }],
    }).finally(() => {
      unsub()
    })

    // 6. 收集结果
    const messages = await Session.messages({ sessionID: session.id })
    const summary = messages
      .filter((x) => x.info.role === "assistant")
      .flatMap((msg) => msg.parts.filter((x) => x.type === "tool"))

    const text = result.parts.findLast((x) => x.type === "text")?.text ?? ""

    return {
      title: params.description,
      metadata: { summary, sessionId: session.id },
      output: text + "\n\n" + `<task_metadata>\nsession_id: ${session.id}\n</task_metadata>`,
    }
  },
})
```

## 输出截断处理

当工具输出过大时，需要进行截断处理以避免超过上下文限制。

```typescript
export const Truncate = {
  async GLOB: "**/.opencode/output/truncated-*.txt",

  async output(input: string, options: {}, agent?: Agent.Info) {
    const maxLength = agent?.truncate?.maxLength ?? TRUNCATE_DEFAULT_LENGTH

    if (input.length <= maxLength) {
      return { content: input, truncated: false }
    }

    // 1. 创建截断文件
    const filename = `truncated-${Date.now()}.txt`
    const outputPath = path.join(Instance.worktree, ".opencode", "output", filename)

    await fs.ensureDir(path.dirname(outputPath))
    await Bun.write(outputPath, input)

    // 2. 生成截断消息
    const truncated = input.slice(0, maxLength)
    const message = `[Output was truncated to ${maxLength} characters. Full output saved to ${outputPath}]`

    return {
      content: truncated + "\n\n" + message,
      truncated: true,
      outputPath,
    }
  },
}
```
