    // 订阅所有事件

      await stream.writeSSE({
        data: JSON.stringify(event),
      })

        data: JSON.stringify({ type: "server.heartbeat" }),
      })

        unsub()
        resolve()
      })
    })
  })

// SDK 客户端
async function* subscribe() {
  const response = await fetch(`${baseUrl}/event`)
  const reader = response.body.getReader()
  const decoder = new TextDecoder()

  while (true) {
    const { done, value } = await reader.read()
    if (done) break

    const text = decoder.decode(value)
    const lines = text.split("\n")

    for (const line of lines) {
      if (line.startsWith("data: ")) {
        const event = JSON.parse(line.slice(6))
        yield event
      }
    }
  }
}

## 专题三：任务拆分技术

### 3.1 Token 限制与截断策略

#### 输出截断

```typescript
// src/tool/truncation.ts
export async function output(
  input: string,
  options: {},
  agent?: Agent.Info
) {
  const maxLength = agent?.truncate?.maxLength ?? TRUNCATE_DEFAULT_LENGTH

  if (input.length <= maxLength) {
    return { content: input, truncated: false }
  }

  // 1. 将完整输出保存到文件
  const filename = `truncated-${Date.now()}.txt`
  const outputPath = path.join(
    Instance.worktree,
    ".opencode",
    "output",
    filename
  )
  await Bun.write(outputPath, input)

  // 2. 返回截断后的内容和路径
  const truncated = input.slice(0, maxLength)
  const message = `[Output was truncated. Full output saved to ${outputPath}]`

  return {
    content: truncated + "\n\n" + message,
    truncated: true,
    outputPath,
  }
}
```

#### 截断后的后果与实际机制

**截断后 OpenCode 做了什么：**

OpenCode 执行以下操作：
1. 保存完整输出到文件系统
2. 返回截断后的内容 + 提示信息

```typescript
// src/tool/truncation.ts:96-104
const hint = `The tool call succeeded but the output was truncated.
              Full output saved to: ${filepath}
              Use Grep to search the full content or Read with offset/limit.`

const message = `${preview}\n\n...${removed} ${unit} truncated...\n\n${hint}`
```

**关键：OpenCode 只是返回了一个提示，并没有自动做任何补救操作。**

**大模型实际收到的信息：**

```
[前 2000 行内容]

...1000 lines truncated...

The tool call succeeded but the output was truncated.
Full output saved to: /path/to/tool-output/tool_abc123
Use Grep to search the full content or Read with offset/limit to view specific sections.
```

**截断的后果：**

1. **信息不完整**
   - 大模型只能看到截断部分的内容
   - 截断部分的内容对大模型来说是"未知"的
   - 如果关键信息在截断部分，大模型无法知道

2. **大模型需要自己判断**
   - OpenCode 不会自动采取任何行动
   - 大模型需要自己决定是否需要获取完整信息
   - 这个判断可能正确，也可能错误

3. **分段读取的固有限制**
   - 使用 Grep：只能看到匹配的行，缺少上下文
   - 使用 Read 带参数：只能看到指定范围的内容
   - 需要多轮调用才能覆盖完整文件
   - 每轮调用都会消耗 Token

**大模型的实际处理（完全依赖大模型自己的判断）：**

可能的情况 A：忽略提示
```
大模型："这 2000 行内容已经足够了"
→ 不做任何额外操作
→ 可能遗漏截断部分的关键信息
```

可能的情况 B：尝试搜索
```
大模型："我在找错误，但截断了，可能在后面"
→ Grep({ pattern: "error", path: "完整文件路径" })
→ 如果找到，继续处理
→ 如果没找到，可能还是遗漏了
```

可能的情况 C：委托给子代理
```
大模型："完整内容太长，让 explore 子代理处理"
→ Task({
    subagent_type: "explore",
    prompt: "分析这个文件，找出所有关键信息"
  })
→ 子代理在独立的子会话中处理完整文件
→ 子代理的 Token 使用不影响主会话
→ 返回汇总结果
```

**为什么会有子代理机制？**

正是因为截断会丢失信息，所以 OpenCode 提供了 Task 工具：

```
主会话的限制：
- Token 总量有限（如 200,000）
- 工具输出截断后信息不完整
- 大模型基于不完整信息做判断

子会话的优势：
- 独立的 Token 限制
- 子代理可以多次调用工具处理完整文件
  Read({ offset: 0, limit: 500 })
  Read({ offset: 500, limit: 500 })
  Read({ offset: 1000, limit: 500 })
- 这些调用的 Token 消耗在子会话中
- 最终只返回汇总结果给主会话
- 主会话的 Token 使用量很小
```

**总结：**

截断是一个权衡机制，没有完美的解决方案：

| 优点 | 缺点/风险 |
|------|----------|
| 避免单次工具调用超过 Token 限制 | 大模型只能看到部分内容 |
| 保存完整内容，允许后续访问 | 大模型可能忽略提示，不访问完整内容 |
| 提供多种访问方式（Grep/Read/Task） | 即使访问，也可能是基于不完整的理解 |
| 子代理可以独立处理大文件 | 子代理的判断也可能有误 |

**关键理解：OpenCode 提供了工具（保存文件、提示路径），但真正的"智能"来自大模型自己的判断，而这个判断是有局限性的。**

---

**代码文件截断的特殊问题**

代码文件与日志文件完全不同：

```
日志文件（前重后轻）：
前 2000 行：主要日志内容
后 1000 行：新增日志（可能是重复的）

代码文件（前轻后重）：
前 2000 行：import、注释、辅助函数
后 1000 行：核心类定义 ← 截断会丢失核心逻辑！
```

**如果代码文件被截断，大模型可能完全不知道关键方法的存在。**

**OpenCode 如何解决这个问题？**

详细分析请参阅：`zh/16-代码库理解机制.md`

**简要说明：**

OpenCode 不是靠"截断后智能恢复"，而是从根本上避免需要读取完整大文件：

1. **定位优先**：先用 Glob + Grep 找文件和关键位置
2. **分段读取**：用 Read 的 offset/limit 参数只读需要的部分
3. **子代理隔离**：复杂分析用 Task 工具创建独立子会话

```
错误做法（会截断）：
Read({ filePath: "large-file.ts" })  → 读取 10000 行 → 触发截断

正确做法（不截断）：
Grep({ pattern: "class OrderProcessor" })  → 找到行号 150
Read({ filePath: "large-file.ts", offset: 145, limit: 100 })  → 只读类定义部分
```

#### 会话压缩

```typescript
// src/session/compaction.ts
export async function isOverflow(input: {
  tokens: MessageV2.Tokens
  model: Provider.Model
}) {
  const inputTokens = input.tokens.input
  const threshold = input.model.limit.input * COMPACT_RATIO

  return inputTokens > threshold
}

// 当超过阈值时，触发压缩
if (await SessionCompaction.isOverflow({ tokens: usage.tokens, model: input.model })) {
  needsCompaction = true
}

// 压缩通过移除旧工具结果的完整内容实现
async function compact() {
  for (const part of parts) {
    if (part.type === "tool" && part.state.status === "completed") {
      await Session.updatePart({
        ...part,
        state: {
          ...part.state,
          output: "[Old tool result content cleared]",
          time: { ...part.state.time, compacted: Date.now() },
        },
      })
    }
  }
}
```

### 3.2 子代理任务分解

#### Task 工具的工作原理

```typescript
// src/tool/task.ts
execute: async (params, ctx) => {
  // 1. 创建子会话
  const session = await Session.create({
    parentID: ctx.sessionID,  // 父会话 ID
    title: params.description + ` (@${agent.name} subagent)`,
    permission: [/* 限制权限 */],
  })

  // 2. 监听子会话事件
  const unsub = Bus.subscribe(MessageV2.Event.PartUpdated, async (evt) => {
    if (evt.properties.part.sessionID !== session.id) return
    // 更新父会话中的元数据
    ctx.metadata({ title: params.description, metadata: { ... } })
  })

  // 3. 发送任务给子代理
  const result = await SessionPrompt.prompt({
    messageID,
    sessionID: session.id,  // 子会话
    model: agent.model ?? { /* 继承父会话模型 */ },
    agent: agent.name,      // 子代理名称
    tools: { /* 禁用某些工具 */ },
    parts: [{ type: "text", text: params.prompt }],
  })

  // 4. 收集结果
  const messages = await Session.messages({ sessionID: session.id })
  const summary = extractToolSummary(messages)

  return {
    title: params.description,
    metadata: { summary, sessionId: session.id },
    output: formatOutput(result, summary),
  }
}
```

### 3.3 大代码库处理策略

**注意：本节内容已移至专门的详细文档**
- 请查看 `zh/15-大代码库处理策略详解.md` 获取完整的处理流程

**快速概览**

OpenCode 使用五种策略处理大型代码库：

1. **输出截断** - 单个工具输出超过限制时，保存到文件并返回摘要
2. **会话压缩** - Token 溢出时，LLM 生成对话摘要
3. **会话剪枝** - 自动清除旧工具结果内容
4. **子代理分解** - 使用 Task 工具创建子会话处理复杂任务
5. **批量并行** - 使用 Batch 工具并行执行多个工具调用

详细触发条件、执行流程和源代码分析，请参阅 `zh/15-大代码库处理策略详解.md`。
